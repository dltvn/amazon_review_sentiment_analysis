inputs = tokenizer(
    prompt,
    return_tensors="pt",
    truncation=True,
    max_length=512
)


outputs = model.generate(
    **inputs,
    max_new_tokens=120,
    num_beams=4,
    temperature=0.7,
    early_stopping=True
)

generated_response = tokenizer.decode(outputs[0], skip_special_tokens=True)

print("Generated service response:\n")
print(generated_response)
